# functions to apply Red, Blue, or Green filter to desired photo 
def redFilter(image to apply filter)
{
    with Image.open("/file path/image name.jpg") as im:
        im = im.convert("RGB")

    redImage = im.new("RGB", image.size, (255,0,0)) # tupple for red color
    filRed_image = im.blend(im, redImage, 0.5)    # apply red to whole picture

    filRed_image.save("/save file path/image_red_.jpg")

}
